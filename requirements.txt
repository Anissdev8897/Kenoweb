# Configuration Render pour Keno Analyzer Pro
services:
  - type: web
    name: keno-analyzer-pro
    env: python
    buildCommand: |
      pip install --upgrade pip setuptools wheel && pip install -r requirements.txt
    startCommand: gunicorn -w 4 -b 0.0.0.0:$PORT main:app
    envVars:
      - key: PYTHON_VERSION
        value: "3.10"
      - key: DATABASE_URL
        fromDatabase:
          name: keno-db
          property: connectionString
      - key: FLASK_ENV
        value: production
      - key: PYTHONUNBUFFERED
        value: "1"
      - key: SECRET_KEY
        sync: false
      - key: OPENAI_API_KEY
        sync: false
    disk:
      name: keno-data
      mountPath: /app/data
      sizeGB: 10

databases:
  - name: keno-db
    databaseName: keno_analyzer
    user: keno_user
    plan: starter
    postgresExtensions:
      - uuid-ossp
      - pgcrypto
